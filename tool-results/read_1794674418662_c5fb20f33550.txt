   695→function project(d, sx, sy, cam){
   696→  // returns {x,y,factor,vis} for a waypoint at depth d, scatter (sx,sy)
   697→  var tz = cam - d;
   698→  var denom = PERSP - tz;
   699→  if(denom < 1) return {x:cx,y:cy,factor:0,tz:tz,vis:0};
   700→  var factor = PERSP/denom;
   701→  return {x:cx+sx*factor, y:cy+sy*factor, factor:factor, tz:tz};
   702→}
   703→function tzOpacity(tz){
   704→  // fade in from far, full near camera, fade out after passing
   705→  if(tz >= PERSP*0.85) return 0;            // behind camera
   706→  if(tz <= -PERSP*5)    return 0;            // too far ahead
   707→  if(tz <= -PERSP*2.5)  return (tz + PERSP*5)/(PERSP*2.5);   // 0..1 ramp in
   708→  if(tz <= PERSP*0.15)  return 1;            // full
   709→  return 1 - (tz - PERSP*0.15)/(PERSP*0.7);  // fade out
   710→}

   715→  if(targetDepth<=wp[0].depth) return {x:wp[0].x,y:wp[0].y,d:wp[0].depth};
   716→  if(targetDepth>=wp[wp.length-1].depth){ var last=wp[wp.length-1]; return {x:last.x,y:last.y,d:last.depth}; }
   717→  for(var i=0;i<wp.length-1;i++){
   718→    if(targetDepth>=wp[i].depth && targetDepth<=wp[i+1].depth){
   719→      var t=(targetDepth-wp[i].depth)/(wp[i+1].depth-wp[i].depth);
   720→      return {x:wp[i].x+(wp[i+1].x-wp[i].x)*t, y:wp[i].y+(wp[i+1].y-wp[i].y)*t, d:targetDepth};
   721→    }
   722→  }
   723→  return {x:0,y:0,d:targetDepth};
   724→}
